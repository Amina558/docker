# Run Verventech website

# 🥇 Pull the Docker Image
Commands = sudo docker pull verventech/verventech-website
<img width="1920" height="553" alt="image" src="https://github.com/user-attachments/assets/c13100f8-673f-43e1-bbfe-62c48e0e4f4d" />

# 🥈Run the container 
commands = sudo docker run -dt --name my-ver-web -p 8081:80 verventech/verventech-website
# ➡️8081----->host port number. put in left side
# ➡️80------->container port number. put in right side
➡️--name my-ver-web: container name
